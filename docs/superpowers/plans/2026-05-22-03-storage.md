# Storage (cr-sqlite + SQLCipher) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `StorageBackend` using cr-sqlite (CRDT-aware SQLite extension) with SQLCipher at-rest encryption. Two DB files: `meta.db` (auth metadata, dataset registry) and one `{app_id}/data.db` per tenant.

**Architecture:** `storage/crsqlite/` package wraps `mattn/go-sqlite3` with CGo extension loading. cr-sqlite extension binary is downloaded separately and path configured via env. The package exposes a single `Open()` constructor that returns a `core.StorageBackend`.

**Tech Stack:** Go 1.22+, github.com/mattn/go-sqlite3 (CGo), cr-sqlite extension (.so/.dylib from vlcn-io/cr-sqlite releases), SQLCipher (via build tag or separate driver).

**Prerequisite:** Plan 01 complete. cr-sqlite extension binary available at `$CRSQLITE_EXT_PATH`.

**Note on SQLCipher + cr-sqlite:** SQLCipher is a patched SQLite build. cr-sqlite is a runtime-loadable extension. They are compatible: open the DB with SQLCipher driver, then load the cr-sqlite extension. The CGo build tag `sqlite_unlock_notify` enables extension loading in go-sqlite3.

```go
// storage/crsqlite/open.go
package crsqlite

import (
	"database/sql"
	"fmt"

	_ "github.com/mattn/go-sqlite3"
)

// Handle wraps a SQLite database connection with cr-sqlite loaded.
type Handle struct {
	db *sql.DB
}

// Open opens (or creates) a SQLCipher-encrypted SQLite database and loads
// the cr-sqlite extension. key must be exactly 32 bytes (raw, not hex).
// extPath is the filesystem path to the cr-sqlite .so/.dylib extension.
func Open(path, key, extPath string) (*Handle, error) {
	dsn := fmt.Sprintf(
		"file:%s?_pragma=key('%s')&_pragma=journal_mode(WAL)&_pragma=foreign_keys(ON)",
		path, key,
	)
	db, err := sql.Open("sqlite3", dsn)
	if err != nil {
		return nil, fmt.Errorf("open db: %w", err)
	}
	db.SetMaxOpenConns(1) // SQLite: single writer

	conn, err := db.Conn(nil)
	if err != nil {
		db.Close()
		return nil, fmt.Errorf("acquire conn: %w", err)
	}
	defer conn.Close()

	_, err = conn.ExecContext(nil, fmt.Sprintf(`SELECT load_extension('%s')`, extPath))
	if err != nil {
		db.Close()
		return nil, fmt.Errorf("load cr-sqlite extension: %w", err)
	}

	if err := db.Ping(); err != nil {
		db.Close()
		return nil, fmt.Errorf("ping after extension load: %w", err)
	}
	return &Handle{db: db}, nil
}

// DB returns the underlying *sql.DB for schema init and queries.
func (h *Handle) DB() *sql.DB { return h.db }

// Close closes the database connection.
func (h *Handle) Close() error { return h.db.Close() }
```

```go
package crsqlite

import "fmt"

const metaSchema = `
CREATE TABLE IF NOT EXISTS device_keys (
    device_key_id  TEXT PRIMARY KEY,
    device_id      TEXT NOT NULL,
    app_id         TEXT NOT NULL,
    key_hash       TEXT NOT NULL,
    status         TEXT NOT NULL DEFAULT 'active',
    registered_at  DATETIME NOT NULL DEFAULT (datetime('now')),
    last_seen_at   DATETIME
);

CREATE TABLE IF NOT EXISTS datasets (
    app_id         TEXT NOT NULL,
    dataset_id     TEXT NOT NULL,
    acl            TEXT NOT NULL DEFAULT 'private',
    schema_version INTEGER NOT NULL DEFAULT 1,
    PRIMARY KEY (app_id, dataset_id)
);

CREATE TABLE IF NOT EXISTS refresh_tokens (
    token_hash TEXT PRIMARY KEY,
    app_id     TEXT NOT NULL,
    user_id    TEXT NOT NULL,
    expires_at DATETIME NOT NULL,
    used       INTEGER NOT NULL DEFAULT 0
);

CREATE INDEX IF NOT EXISTS idx_device_keys_app ON device_keys(app_id);
CREATE INDEX IF NOT EXISTS idx_device_keys_device ON device_keys(device_id);
CREATE INDEX IF NOT EXISTS idx_refresh_tokens_user ON refresh_tokens(app_id, user_id);
`

// MetaDB wraps the meta.db connection (auth + dataset registry).
type MetaDB struct {
	*Handle
}

// OpenMeta opens meta.db and applies the schema.
func OpenMeta(path, key, extPath string) (*MetaDB, error) {
	h, err := Open(path, key, extPath)
	if err != nil {
		return nil, err
	}
	if _, err := h.db.Exec(metaSchema); err != nil {
		h.Close()
		return nil, fmt.Errorf("apply meta schema: %w", err)
	}
	return &MetaDB{h}, nil
}
```

```go
// storage/crsqlite/data.go
package crsqlite

import (
	"fmt"
	"regexp"
)

var safeIDRe = regexp.MustCompile(`^[a-zA-Z0-9_-]+$`)

// DataDB wraps the per-app data.db connection.
type DataDB struct {
	*Handle
}

// OpenData opens (or creates) a per-app data.db and applies base schema.
func OpenData(path, key, extPath string) (*DataDB, error) {
	h, err := Open(path, key, extPath)
	if err != nil {
		return nil, err
	}
	// Base tables shared across all datasets
	const base = `
CREATE TABLE IF NOT EXISTS snapshots (
    dataset_id  TEXT NOT NULL,
    user_id     TEXT NOT NULL,
    doc_id      TEXT NOT NULL,
    data        TEXT NOT NULL,
    wall_time   INTEGER NOT NULL,
    logical     INTEGER NOT NULL,
    device_id   TEXT NOT NULL,
    created_at  DATETIME NOT NULL DEFAULT (datetime('now')),
    PRIMARY KEY (dataset_id, user_id, doc_id)
);`
	if _, err := h.db.Exec(base); err != nil {
		h.Close()
		return nil, fmt.Errorf("apply data base schema: %w", err)
	}
	return &DataDB{h}, nil
}

// EnsureDataset creates the op_log table for a dataset if it does not exist.
// dataset_id must be alphanumeric/underscore/dash.
func (d *DataDB) EnsureDataset(datasetID string) error {
	if !safeIDRe.MatchString(datasetID) {
		return fmt.Errorf("invalid dataset_id: %q", datasetID)
	}
	ddl := fmt.Sprintf(`
CREATE TABLE IF NOT EXISTS op_log_%s (
    op_id      TEXT PRIMARY KEY,
    user_id    TEXT NOT NULL,
    doc_id     TEXT NOT NULL,
    field      TEXT NOT NULL,
    value      TEXT NOT NULL,
    wall_time  INTEGER NOT NULL,
    logical    INTEGER NOT NULL,
    device_id  TEXT NOT NULL,
    status     TEXT NOT NULL DEFAULT 'active'
);
CREATE INDEX IF NOT EXISTS idx_op_log_%s_clock ON op_log_%s(user_id, wall_time, logical, device_id);
SELECT crsql_as_crr('op_log_%s');
`, datasetID, datasetID, datasetID, datasetID)
	_, err := d.db.Exec(ddl)
	return err
}
```

```go
// storage/crsqlite/backend.go
package crsqlite

import (
	"context"
	"database/sql"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"sync"

	"github.com/yourusername/mobile-db/core"
)

// Backend implements core.StorageBackend using cr-sqlite + SQLCipher.
type Backend struct {
	meta    *MetaDB
	dataDir string
	key     string
	extPath string
	mu      sync.RWMutex
	apps    map[string]*DataDB // app_id → data db
}

// NewBackend opens meta.db and prepares the data directory.
func NewBackend(metaPath, dataDir, key, extPath string) (*Backend, error) {
	if err := os.MkdirAll(dataDir, 0700); err != nil {
		return nil, err
	}
	meta, err := OpenMeta(metaPath, key, extPath)
	if err != nil {
		return nil, err
	}
	return &Backend{
		meta:    meta,
		dataDir: dataDir,
		key:     key,
		extPath: extPath,
		apps:    map[string]*DataDB{},
	}, nil
}

func (b *Backend) appDB(appID string) (*DataDB, error) {
	b.mu.RLock()
	d, ok := b.apps[appID]
	b.mu.RUnlock()
	if ok {
		return d, nil
	}
	b.mu.Lock()
	defer b.mu.Unlock()
	if d, ok = b.apps[appID]; ok {
		return d, nil
	}
	dir := filepath.Join(b.dataDir, appID)
	if err := os.MkdirAll(dir, 0700); err != nil {
		return nil, err
	}
	d, err := OpenData(filepath.Join(dir, "data.db"), b.key, b.extPath)
	if err != nil {
		return nil, err
	}
	b.apps[appID] = d
	return d, nil
}

func (b *Backend) MergeOps(ctx context.Context, appID, userID, datasetID string, ops []core.CRDTOp) error {
	d, err := b.appDB(appID)
	if err != nil {
		return err
	}
	if err := d.EnsureDataset(datasetID); err != nil {
		return err
	}
	table := "op_log_" + datasetID
	stmt := fmt.Sprintf(
		`INSERT OR REPLACE INTO %s (op_id, user_id, doc_id, field, value, wall_time, logical, device_id)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?)`, table)
	tx, err := d.db.BeginTx(ctx, nil)
	if err != nil {
		return err
	}
	defer tx.Rollback()
	for _, op := range ops {
		v, err := json.Marshal(op.Value)
		if err != nil {
			return err
		}
		if _, err := tx.ExecContext(ctx, stmt,
			op.OpID, userID, op.DocID, op.Field, string(v),
			op.Timestamp.WallTime, op.Timestamp.Logical, op.Timestamp.DeviceID,
		); err != nil {
			return err
		}
	}
	return tx.Commit()
}

func (b *Backend) GetDelta(ctx context.Context, appID, userID, datasetID string, since core.HLC) ([]core.CRDTOp, core.HLC, error) {
	d, err := b.appDB(appID)
	if err != nil {
		return nil, core.HLC{}, err
	}
	table := "op_log_" + datasetID
	query := fmt.Sprintf(`
		SELECT op_id, doc_id, field, value, wall_time, logical, device_id
		FROM %s
		WHERE user_id = ?
		  AND status = 'active'
		  AND (wall_time > ? OR (wall_time = ? AND logical > ?) OR (wall_time = ? AND logical = ? AND device_id > ?))
		ORDER BY wall_time, logical, device_id`, table)
	rows, err := d.db.QueryContext(ctx, query,
		userID,
		since.WallTime, since.WallTime, since.Logical,
		since.WallTime, since.Logical, since.DeviceID)
	if err != nil {
		return nil, core.HLC{}, err
	}
	defer rows.Close()

	var ops []core.CRDTOp
	var maxClock core.HLC
	for rows.Next() {
		var op core.CRDTOp
		var valueJSON string
		if err := rows.Scan(&op.OpID, &op.DocID, &op.Field, &valueJSON,
			&op.Timestamp.WallTime, &op.Timestamp.Logical, &op.Timestamp.DeviceID); err != nil {
			return nil, core.HLC{}, err
		}
		if err := json.Unmarshal([]byte(valueJSON), &op.Value); err != nil {
			return nil, core.HLC{}, err
		}
		op.DeviceID = op.Timestamp.DeviceID
		ops = append(ops, op)
		maxClock = op.Timestamp
	}
	return ops, maxClock, rows.Err()
}

func (b *Backend) OpCount(ctx context.Context, appID, userID, datasetID string) (int, error) {
	d, err := b.appDB(appID)
	if err != nil {
		return 0, err
	}
	var count int
	err = d.db.QueryRowContext(ctx,
		fmt.Sprintf(`SELECT COUNT(*) FROM op_log_%s WHERE user_id = ? AND status = 'active'`, datasetID),
		userID).Scan(&count)
	return count, err
}

func (b *Backend) SchemaVersion(ctx context.Context, appID, datasetID string) (int, error) {
	var version int
	err := b.meta.db.QueryRowContext(ctx,
		`SELECT schema_version FROM datasets WHERE app_id = ? AND dataset_id = ?`,
		appID, datasetID).Scan(&version)
	if err == sql.ErrNoRows {
		return 1, nil
	}
	return version, err
}

func (b *Backend) GetSnapshot(ctx context.Context, appID, userID, datasetID string) (map[string]any, core.HLC, error) {
	d, err := b.appDB(appID)
	if err != nil {
		return nil, core.HLC{}, err
	}
	rows, err := d.db.QueryContext(ctx,
		`SELECT doc_id, data, wall_time, logical, device_id FROM snapshots
		 WHERE dataset_id = ? AND user_id = ?`, datasetID, userID)
	if err != nil {
		return nil, core.HLC{}, err
	}
	defer rows.Close()

	docs := map[string]any{}
	var maxClock core.HLC
	for rows.Next() {
		var docID, dataStr string
		var clk core.HLC
		if err := rows.Scan(&docID, &dataStr, &clk.WallTime, &clk.Logical, &clk.DeviceID); err != nil {
			return nil, core.HLC{}, err
		}
		var doc any
		if err := json.Unmarshal([]byte(dataStr), &doc); err != nil {
			return nil, core.HLC{}, err
		}
		docs[docID] = doc
		maxClock = clk
	}
	return docs, maxClock, rows.Err()
}

func (b *Backend) Compact(ctx context.Context, appID, userID, datasetID string) error {
	d, err := b.appDB(appID)
	if err != nil {
		return err
	}
	// Materialize current state for each doc
	ops_q := fmt.Sprintf(`
		SELECT op_id, doc_id, field, value, wall_time, logical, device_id
		FROM op_log_%s WHERE user_id = ? AND status = 'active'`, datasetID)
	rows, err := d.db.QueryContext(ctx, ops_q, userID)
	if err != nil {
		return err
	}
	defer rows.Close()

	var allOps []core.CRDTOp
	for rows.Next() {
		var op core.CRDTOp
		var valueJSON string
		if err := rows.Scan(&op.OpID, &op.DocID, &op.Field, &valueJSON,
			&op.Timestamp.WallTime, &op.Timestamp.Logical, &op.Timestamp.DeviceID); err != nil {
			return err
		}
		json.Unmarshal([]byte(valueJSON), &op.Value)
		allOps = append(allOps, op)
	}
	rows.Close()

	// Use core merge to get current state per doc
	merged := mergeOpsLocal(allOps)

	tx, err := d.db.BeginTx(ctx, nil)
	if err != nil {
		return err
	}
	defer tx.Rollback()

	var maxWall int64
	var maxLogical uint32
	var maxDevice string
	for _, op := range allOps {
		if op.Timestamp.WallTime > maxWall ||
			(op.Timestamp.WallTime == maxWall && op.Timestamp.Logical > maxLogical) {
			maxWall, maxLogical, maxDevice = op.Timestamp.WallTime, op.Timestamp.Logical, op.Timestamp.DeviceID
		}
	}

	for docID, fields := range merged {
		data, _ := json.Marshal(fields)
		_, err := tx.ExecContext(ctx,
			`INSERT OR REPLACE INTO snapshots (dataset_id, user_id, doc_id, data, wall_time, logical, device_id)
			 VALUES (?, ?, ?, ?, ?, ?, ?)`,
			datasetID, userID, docID, string(data), maxWall, maxLogical, maxDevice)
		if err != nil {
			return err
		}
	}

	// Archive compacted ops
	_, err = tx.ExecContext(ctx,
		fmt.Sprintf(`UPDATE op_log_%s SET status = 'archived' WHERE user_id = ? AND status = 'active'`, datasetID),
		userID)
	if err != nil {
		return err
	}
	return tx.Commit()
}

func (b *Backend) Close() error {
	b.mu.Lock()
	defer b.mu.Unlock()
	for _, d := range b.apps {
		d.Close()
	}
	return b.meta.Close()
}

// mergeOpsLocal is a package-local copy of core/crdt.Merge to avoid circular imports.
func mergeOpsLocal(ops []core.CRDTOp) map[string]map[string]any {
	type win struct{ op core.CRDTOp }
	wins := map[string]map[string]win{}
	for _, op := range ops {
		if _, ok := wins[op.DocID]; !ok {
			wins[op.DocID] = map[string]win{}
		}
		cur, exists := wins[op.DocID][op.Field]
		if !exists || cur.op.Timestamp.WallTime < op.Timestamp.WallTime ||
			(cur.op.Timestamp.WallTime == op.Timestamp.WallTime && cur.op.Timestamp.Logical < op.Timestamp.Logical) {
			wins[op.DocID][op.Field] = win{op}
		}
	}
	result := map[string]map[string]any{}
	for docID, fields := range wins {
		result[docID] = map[string]any{}
		for f, w := range fields {
			result[docID][f] = w.op.Value
		}
	}
	return result
}
```
